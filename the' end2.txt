    // Create an array of materials, one for each face, using the canvas texture
    // Order: [+x, -x, +y, -y, +z, -z] (right, left, top, bottom, front, back)
    // Standard die layout: 1-6, 2-5, 3-4 are opposite
    const diceMaterials = [
        new THREE.MeshPhongMaterial({ map: createDiceFaceTexture(3), color: 0xffffff, specular: 0x111111, shininess: 10 }), // right
        new THREE.MeshPhongMaterial({ map: createDiceFaceTexture(4), color: 0xffffff, specular: 0x111111, shininess: 10 }), // left
        new THREE.MeshPhongMaterial({ map: createDiceFaceTexture(5), color: 0xffffff, specular: 0x111111, shininess: 10 }), // top
        new THREE.MeshPhongMaterial({ map: createDiceFaceTexture(2), color: 0xffffff, specular: 0x111111, shininess: 10 }), // bottom
        new THREE.MeshPhongMaterial({ map: createDiceFaceTexture(1), color: 0xffffff, specular: 0x111111, shininess: 10 }), // front
        new THREE.MeshPhongMaterial({ map: createDiceFaceTexture(6), color: 0xffffff, specular: 0x111111, shininess: 10 })  // back
    ];

    diceMesh1 = new THREE.Mesh(diceGeometry, diceMaterials);
    diceMesh2 = new THREE.Mesh(diceGeometry, diceMaterials);

    diceMesh1.position.x = -DICE_SIZE * 0.75;
    diceMesh2.position.x = DICE_SIZE * 0.75;

    scene.add(diceMesh1);
    scene.add(diceMesh2);

    // Initial neutral rotation
    diceMesh1.rotation.set(0.2, 0.3, 0.1);
    diceMesh2.rotation.set(-0.1, 0.2, -0.3);

    // Animate loop
    renderer.setAnimationLoop(animateDice);
};

const animateDice = () => {
    renderer.render(scene, camera);
};

const roll3DDice = (die1, die2, onComplete) => {
    // Reset position and rotation to simulate a new throw
    gsap.set(diceMesh1.rotation, { x: 0, y: 0, z: 0 });
    gsap.set(diceMesh2.rotation, { x: 0, y: 0, z: 0 });
    gsap.set(diceMesh1.position, { y: 0 });
    gsap.set(diceMesh2.position, { y: 0 });

    const DURATION = 1.5;
    const TOTAL_ROTATION = 10; // rotations

    // GSAP Timeline for Die 1
    gsap.timeline()
        .to(diceMesh1.position, { duration: DURATION / 3, y: 40, ease: "power1.out" }) // Jump up
        .to(diceMesh1.position, { duration: DURATION * 2 / 3, y: 0, ease: "bounce.out" }, `-=${DURATION * 2 / 3}`) // Fall down and bounce
        .to(diceMesh1.rotation, { 
            duration: DURATION,
            x: `+=${Math.random() * TOTAL_ROTATION + TOTAL_ROTATION * 2}*${Math.PI}`,
            y: `+=${Math.random() * TOTAL_ROTATION + TOTAL_ROTATION * 2}*${Math.PI}`,
            z: `+=${Math.random() * TOTAL_ROTATION + TOTAL_ROTATION * 2}*${Math.PI}`,
            ease: "power2.out",
            onComplete: () => {
                // Find final rotation to represent the dice value
                const finalRotation = getDiceRotation(die1);
                gsap.to(diceMesh1.rotation, { 
                    duration: 0.7, // Increased duration for a clearer "snap"
                    ...finalRotation, 
                    ease: "back.out(1.7)" // Use a more pronounced ease
                });
            }
        }, 0); // Start rotation at the beginning

    // GSAP Timeline for Die 2
    gsap.timeline({
        onComplete: onComplete // Call original completion handler after second die lands
    })
        .to(diceMesh2.position, { duration: DURATION / 3, y: 45, ease: "power1.out", delay: 0.1 }) // Slight delay for second die
        .to(diceMesh2.position, { duration: DURATION * 2 / 3, y: 0, ease: "bounce.out" }, `-=${DURATION * 2 / 3}`)
        .to(diceMesh2.rotation, { 
            duration: DURATION,
            x: `+=${Math.random() * TOTAL_ROTATION + TOTAL_ROTATION * 2}*${Math.PI}`,
            y: `+=${Math.random() * TOTAL_ROTATION + TOTAL_ROTATION * 2}*${Math.PI}`,
            z: `+=${Math.random() * TOTAL_ROTATION + TOTAL_ROTATION * 2}*${Math.PI}`,
            ease: "power2.out",
            onComplete: () => {
                const finalRotation = getDiceRotation(die2);
                gsap.to(diceMesh2.rotation, { 
                    duration: 0.7, // Increased duration
                    ...finalRotation, 
                    ease: "back.out(1.7)" // Use a more pronounced ease
                });
            }
        }, 0.1);
};

// Helper to determine a good resting rotation for a given face value (simplified visual mapping)
const getDiceRotation = (value) => {
    // These rotations move the correct face to the top (+Y direction)
    // Based on the texture order: [+x, -x, +y, -y, +z, -z] => [3, 4, 5, 2, 1, 6]
    const rotations = {
        1: { x: -Math.PI / 2, y: 0, z: 0 }, // Front face (+z) moves to top
        2: { x: Math.PI, y: 0, z: 0 },      // Bottom face (-y) moves to top
        3: { x: 0, y: 0, z: -Math.PI / 2 }, // Right face (+x) moves to top
        4: { x: 0, y: 0, z: Math.PI / 2 },  // Left face (-x) moves to top
        5: { x: 0, y: 0, z: 0 },            // Top face (+y) is already top
        6: { x: Math.PI / 2, y: 0, z: 0 }   // Back face (-z) moves to top
    };
    return rotations[value] || rotations[5]; // Default to 5 if something is wrong
};

/**
 * Creates a canvas texture for a single face of a die.
 * @param {number} faceValue - The number to draw on the face (1-6).
 * @returns {THREE.CanvasTexture} A texture object for use in a material.
 */
const createDiceFaceTexture = (faceValue) => {
    const canvas = document.createElement('canvas');
    const context = canvas.getContext('2d');
    const size = 64;
    canvas.width = size;
    canvas.height = size;

    // White background
    context.fillStyle = '#FFFFFF';
    context.fillRect(0, 0, size, size);
    
    // Rounded corners
    context.beginPath();
    context.moveTo(10, 0);
    context.lineTo(size - 10, 0);
    context.quadraticCurveTo(size, 0, size, 10);
    context.lineTo(size, size - 10);
    context.quadraticCurveTo(size, size, size - 10, size);
    context.lineTo(10, size);
    context.quadraticCurveTo(0, size, 0, size - 10);
    context.lineTo(0, 10);
    context.quadraticCurveTo(0, 0, 10, 0);
    context.closePath();
    context.clip();
    
    context.fillStyle = '#FFFFFF';
    context.fillRect(0, 0, size, size);


    // Dot properties
    context.fillStyle = '#000000';
    const dotRadius = 6;
    const center = size / 2;
    const quarter = size / 4;
    const threeQuarter = size * 3 / 4;

    const drawDot = (x, y) => {
        context.beginPath();
        context.arc(x, y, dotRadius, 0, Math.PI * 2);
        context.fill();
    };

    // Draw dots based on face value
    switch (faceValue) {
        case 1:
            drawDot(center, center);
            break;
        case 2:
            drawDot(quarter, quarter);
            drawDot(threeQuarter, threeQuarter);
            break;
        case 3:
            drawDot(quarter, quarter);
            drawDot(center, center);
            drawDot(threeQuarter, threeQuarter);
            break;
        case 4:
            drawDot(quarter, quarter);
            drawDot(threeQuarter, quarter);
            drawDot(quarter, threeQuarter);
            drawDot(threeQuarter, threeQuarter);
            break;
        case 5:
            drawDot(quarter, quarter);
            drawDot(threeQuarter, quarter);
            drawDot(center, center);
            drawDot(quarter, threeQuarter);
            drawDot(threeQuarter, threeQuarter);
            break;
        case 6:
            drawDot(quarter, quarter);
            drawDot(threeQuarter, quarter);
            drawDot(quarter, center);
            drawDot(threeQuarter, center);
            drawDot(quarter, threeQuarter);
            drawDot(threeQuarter, threeQuarter);
            break;
    }

    return new THREE.CanvasTexture(canvas);
};

// Helper data structure to quickly check for monopolies
const MONOPOLY_GROUP_COUNTS = {};
PROPERTIES.forEach(prop => {
    if (prop.type === "PROPERTY") {
        MONOPOLY_GROUP_COUNTS[prop.group] = (MONOPOLY_GROUP_COUNTS[prop.group] || 0) + 1;
    }
});

/**
 * Checks which monopolies a player holds.
 * @param {object} player - The player object.
 * @returns {string[]} An array of group names where the player has a monopoly.
 */
const getPlayerMonopolies = (player) => {
    const playerGroupCounts = {};
    if (!player || !player.properties) return [];
    
    player.properties.forEach(propId => {
        const prop = PROPERTIES[propId];
        if (prop && prop.type === "PROPERTY") {
            playerGroupCounts[prop.group] = (playerGroupCounts[prop.group] || 0) + 1;
        }
    });

    const monopolies = [];
    for (const group in MONOPOLY_GROUP_COUNTS) {
        if (playerGroupCounts[group] === MONOPOLY_GROUP_COUNTS[group]) {
            monopolies.push(group);
        }
    }
    return monopolies;
};


/**
 * Handles the logic when a player buys a property.
 * @param {number} propertyId - The ID of the property.
 * @param {string} playerId - The ID of the player buying.
 * @param {number|null} overridePrice - An optional price to override the property's default price.
 */
const handleBuyProperty = (propertyId, playerId, overridePrice = null) => {
    const prop = PROPERTIES[propertyId];
    const player = players.find(p => p.id === playerId);
    const price = overridePrice !== null ? overridePrice : prop.price;

    if (player.money >= price) {
        player.money -= price;
        prop.owner = playerId;
        player.properties.push(propertyId);
        appendLog(`${player.name} achète ${prop.name} pour ${price}€.`, 'turn');

        const cellEl = document.getElementById(`cell-${propertyId}`);
        if (cellEl) {
            cellEl.style.boxShadow = `inset 0 0 0 4px ${player.color}`;
        }
        
        closeModal();
        renderPlayerCards();
        switchTurn();
    } else {
        // Player cannot afford it
        if (overridePrice !== null) {
            // It was a special offer they couldn't afford
            appendLog(`${player.name} n'a pas les fonds pour l'offre spéciale sur ${prop.name}.`, 'error');
            closeModal();
            switchTurn();
        } else {
            // It was a standard purchase attempt, so it goes to auction
            appendLog(`${player.name} n'a pas assez d'argent pour acheter ${prop.name}. La propriété va aux enchères.`, 'system');
            closeModal();
            startAuction(propertyId, playerId);
        }
    }
};

/**
 * Starts an interactive auction for a property.
 */
const startAuction = (propertyId, decliningPlayerId) => {
    appendLog(`${PROPERTIES[propertyId].name} est mis aux enchères !`, 'system');
    
    const bidders = players.filter(p => p.id !== decliningPlayerId).map(p => p.id);
    
    if (bidders.length < 1) {
        appendLog("Pas assez de joueurs pour une enchère.", 'system');
        closeModal();
        switchTurn();
        return;
    }

    auctionState = {
        propertyId: propertyId,
        bidders: bidders,
        currentPlayerIndex: 0,
        currentBid: 10, // Starting bid
        lastBidderId: null
    };

    showModal('auction');
};

const handlePlaceBid = () => {
    const bidInput = document.getElementById('bidInput');
    const bidAmount = parseInt(bidInput.value, 10);
    const biddingPlayer = players.find(p => p.id === auctionState.bidders[auctionState.currentPlayerIndex]);

    if (isNaN(bidAmount) || bidAmount <= auctionState.currentBid) {
        alert(`Votre enchère doit être supérieure à ${auctionState.currentBid}€.`);
        return;
    }
    if (bidAmount > biddingPlayer.money) {
        alert("Vous n'avez pas assez d'argent pour cette enchère.");
        return;
    }

    auctionState.currentBid = bidAmount;
    auctionState.lastBidderId = biddingPlayer.id;
    
    // Move to next player
    auctionState.currentPlayerIndex = (auctionState.currentPlayerIndex + 1) % auctionState.bidders.length;
    showModal('auction'); // Refresh modal for the next player
};

const handleWithdrawFromAuction = () => {
    const withdrawingPlayerId = auctionState.bidders[auctionState.currentPlayerIndex];
    
    // Remove player from auction
    auctionState.bidders.splice(auctionState.currentPlayerIndex, 1);
    
    if (auctionState.bidders.length <= 1) {
        endAuction();
        return;
    }

    // Adjust index if needed and refresh modal
    if (auctionState.currentPlayerIndex >= auctionState.bidders.length) {
        auctionState.currentPlayerIndex = 0;
    }
    showModal('auction');
};

const endAuction = () => {
    const prop = PROPERTIES[auctionState.propertyId];
    const winnerId = auctionState.lastBidderId;

    if (winnerId && auctionState.bidders.length > 0) {
        const winner = players.find(p => p.id === winnerId);
        winner.money -= auctionState.currentBid;
        prop.owner = winner.id;
        winner.properties.push(auctionState.propertyId);
        
        appendLog(`${winner.name} remporte ${prop.name} pour ${auctionState.currentBid}€ !`, 'turn');
        
        const cellEl = document.getElementById(`cell-${auctionState.propertyId}`);
        if (cellEl) cellEl.style.boxShadow = `inset 0 0 0 4px ${winner.colorText}`;
        
    } else {
        appendLog(`Personne n'a remporté ${prop.name}. La propriété reste à la banque.`, 'system');
    }

    auctionState = null;
    closeModal();
    renderPlayerCards();
    switchTurn();
};

const handleProposeTrade = () => {

    const partner = players.find(p => p.id === partnerId);

    const offerMoney = parseInt(document.getElementById('swal-money-p1').value, 10) || 0;
    const requestMoney = parseInt(document.getElementById('swal-money-p2').value, 10) || 0;

    const offeredProperties = Array.from(document.querySelectorAll('#swal-p1-panel input[type="checkbox"]:checked')).map(cb => parseInt(cb.id.split('-')[3], 10));
    const requestedProperties = Array.from(document.querySelectorAll('#swal-p2-panel input[type="checkbox"]:checked')).map(cb => parseInt(cb.id.split('-')[3], 10));

    if (offerMoney > activePlayer.money || requestMoney > partner.money) {

        return;
    }

    tradeState = {
        fromPlayer: activePlayer,
        toPlayer: partner,
        offer: { money: offerMoney, properties: offeredProperties, requestMoney: requestMoney, requestProperties: requestedProperties }
    };

    Swal.close();
    setTimeout(() => showModal('trade_offer'), 350);
};

const handleAcceptTrade = () => {
    const { fromPlayer, toPlayer, offer } = tradeState;

    // Correctly transfer money
    fromPlayer.money = fromPlayer.money - offer.money + offer.requestMoney;
    toPlayer.money = toPlayer.money + offer.money - offer.requestMoney;

    // Correctly transfer properties and update cell visuals
    offer.properties.forEach(propId => {
        fromPlayer.properties = fromPlayer.properties.filter(id => id !== propId);
        toPlayer.properties.push(propId);
        PROPERTIES[propId].owner = toPlayer.id;
        const cellEl = document.getElementById(`cell-${propId}`);
        if (cellEl) cellEl.style.boxShadow = `inset 0 0 0 4px ${toPlayer.colorText}`;
    });
    offer.requestProperties.forEach(propId => {
        toPlayer.properties = toPlayer.properties.filter(id => id !== propId);
        fromPlayer.properties.push(propId);
        PROPERTIES[propId].owner = fromPlayer.id;
        const cellEl = document.getElementById(`cell-${propId}`);
        if (cellEl) cellEl.style.boxShadow = `inset 0 0 0 4px ${fromPlayer.colorText}`;
    });

    appendLog(`Échange accepté entre ${fromPlayer.name} et ${toPlayer.name}.`, 'system');
    tradeState = null;
    closeModal();
    
    Swal.fire({
        title: 'Succès !',
        text: 'L\'échange a été effectué.',
        icon: 'success',
        timer: 2000,
        showConfirmButton: false,
        toast: true,
        position: 'top-end'
    });

    renderPlayerCards();
};

/**
 * Handles rent payment when a player lands on an owned property.
 */
const payRent = (payer, ownerId, propertyId, roll = 0) => {
    const prop = PROPERTIES[propertyId];
    const owner = players.find(p => p.id === ownerId);
    
    if (prop.mortgaged) {
        appendLog(`${prop.name} est hypothéquée. Pas de loyer à payer.`, 'system');
        return;
    }

    let rentAmount = 0;
    
    switch (prop.type) {
        case "PROPERTY":
            // Rent is based on houses. Index 0 is un-developed rent.
            rentAmount = prop.rent[prop.houses]; 
            break;
        case "RAILROAD":
            const ownerRailroads = owner.properties.filter(id => PROPERTIES[id].group === 'RAILROAD').length;
            rentAmount = prop.rent[ownerRailroads - 1];
            break;
        case "UTILITY":
            const ownerUtilities = owner.properties.filter(id => PROPERTIES[id].group === 'UTILITY').length;
            const multiplier = RENT_MULTIPLIER_UTILITY[ownerUtilities - 1];
            rentAmount = roll * multiplier;
            break;
        default:
            return; // Should not happen
    }

    appendLog(`${payer.name} paie ${rentAmount}€ de loyer à ${owner.name} pour ${prop.name}.`, 'turn');
    
    if (payer.money >= rentAmount) {
        payer.money -= rentAmount;
        owner.money += rentAmount;
        payer.stats.rentPaid += rentAmount;
        owner.stats.moneyEarned += rentAmount;
    } else {
        const debt = rentAmount - payer.money;
        owner.money += payer.money;
        payer.money = 0;
        handleBankruptcy(payer, debt, owner);
        return; // Bankruptcy flow will handle the next steps
    }

    renderPlayerCards();
};

/**
 * Central function executed after a token lands on a new space.
 */
const handleLandOnSpace = (player, spaceId, roll) => {
    const space = PROPERTIES[spaceId];
    const ownerId = space.owner;
    
    messageBox.textContent = `${player.name} est sur ${space.name}.`;

    if (ownerId && ownerId !== player.id) {
        // Owned by another player: Pay Rent
        payRent(player, ownerId, spaceId, roll);
    } else {
        // Action spaces (Tax, Chance, etc.) or unowned property
        switch (space.type) {
            case "PROPERTY":
            case "RAILROAD":
            case "UTILITY":
                if (!ownerId) {
                    showModal('buy', { propertyId: spaceId, playerName: player.name });
                    return; // Stop the turn flow until modal is resolved
                }
                break;
            case "TAX":
                appendLog(`${player.name} paie ${space.fine}€ d'impôt.`, 'turn');
                if (player.money >= space.fine) {
                    player.money -= space.fine;
                    renderPlayerCards();
                } else {
                    const debt = space.fine - player.money;
                    player.money = 0;
                    handleBankruptcy(player, debt, null); // null creditor is the bank
                    return; 
                }
                break;
            case "CHANCE":
            case "COMMUNITY_CHEST":
                handleCardAction(player, space.type);
                return; // Stop the turn flow until card modal is resolved
            case "GO_TO_JAIL":
                player.position = 10; // Move to Jail square
                player.inJail = true;
                appendLog(`${player.name} va directement en PRISON.`, 'error');
                positionToken(player, players.findIndex(t => t.id === player.id));
                break;
            case "JAIL": // Simple visit to jail
            case "GO": // Pass go money handled in moveToken
            case "FREE_PARKING": // Do nothing
            case "RESERVE":
                break;
        }
    }
    
    // If no modal was shown, switch turn immediately
    switchTurn();
};

const handlePayment = (payer, creditor, amount) => {
    if (payer.money >= amount) {
        payer.money -= amount;
        if (creditor) creditor.money += amount;
    } else {
        const debt = amount - payer.money;
        if (creditor) creditor.money += payer.money;
        payer.money = 0;
        handleBankruptcy(payer, debt, creditor);
    }
    renderPlayerCards();
};

const findNearest = (startPos, type) => {
    let nearestId = -1;
    let minDistance = SPACES;

    PROPERTIES.forEach(prop => {
        if (prop.type === type) {
            let distance = (prop.id - startPos + SPACES) % SPACES;
            if (distance < minDistance) {
                minDistance = distance;
                nearestId = prop.id;
            }
        }
    });
    return nearestId;
};

/**
 * Handles the effects of drawing a card.
 */
const handleCardAction = (player, type) => {
    const deck = type === "CHANCE" ? CHANCE_CARDS : COMMUNITY_CHEST_CARDS;
    const card = deck[Math.floor(Math.random() * deck.length)];

    const pileId = type === "CHANCE" ? '#chance-pile' : '#community-chest-pile';
    const pileEl = document.querySelector(pileId);
    
    const flyingCard = document.createElement('div');
    flyingCard.id = 'flying-card-animation'; // Add a unique ID for testing
    flyingCard.className = `card-pile ${type === "CHANCE" ? 'chance-pile' : 'community-chest-pile'}`;
    flyingCard.style.position = 'absolute';
    flyingCard.style.zIndex = '100';
    flyingCard.innerHTML = pileEl.innerHTML;
    document.body.appendChild(flyingCard);

    gsap.timeline({
        onComplete: () => {
            document.body.removeChild(flyingCard);
            
            // Apply card effect AFTER animation
            switch (card.action.type) {
                case "MOVE_TO":
                    player.position = card.action.target;
                     if (card.action.collectGo) {
                        player.money += PASS_GO_MONEY;
                        appendLog(`${player.name} passe par DÉPART et reçoit ${PASS_GO_MONEY}€.`, 'system');
                    }
                    positionToken(player, players.findIndex(t => t.id === player.id));
                    handleLandOnSpace(player, player.position, 0);
                    return; // handleLandOnSpace will call switchTurn
                case "MOVE_TO_PASS_GO":
                    if (player.position > card.action.target) {
                        player.money += PASS_GO_MONEY;
                        appendLog(`${player.name} passe par DÉPART et reçoit ${PASS_GO_MONEY}€.`, 'system');
                    }
                    player.position = card.action.target;
                    positionToken(player, players.findIndex(t => t.id === player.id));
                    handleLandOnSpace(player, player.position, 0);
                    return;
                case "MOVE_BACK":
                    player.position = (player.position - card.action.steps + SPACES) % SPACES;
                    positionToken(player, players.findIndex(t => t.id === player.id));
                    handleLandOnSpace(player, player.position, 0);
                    return;
                case "PAY":
                    handlePayment(player, null, card.action.amount);
                    break;
                case "COLLECT":
                    player.money += card.action.amount;
                    break;
                case "GO_TO_JAIL":
                    player.position = 10;
                    player.inJail = true;
                    appendLog(`${player.name} va directement en PRISON.`, 'error');
                    positionToken(player, players.findIndex(t => t.id === player.id));
                    break;
                case "COLLECT_FROM_PLAYERS":
                    players.forEach(p => {
                        if (p.id !== player.id) {
                            handlePayment(p, player, card.action.amount);
                        }
                    });
                    break;
                case "REPAIRS":
                    const houseCost = player.properties.reduce((acc, propId) => {
                        const prop = PROPERTIES[propId];
                        if (prop.houses > 0 && prop.houses < 5) {
                            return acc + prop.houses * card.action.house;
                        }
                        return acc;
                    }, 0);
                    const hotelCost = player.properties.reduce((acc, propId) => {
                        const prop = PROPERTIES[propId];
                        if (prop.houses === 5) { // 5 houses = 1 hotel
                            return acc + card.action.hotel;
                        }
                        return acc;
                    }, 0);
                    handlePayment(player, null, houseCost + hotelCost);
                    break;
                case "GET_OUT_OF_JAIL_FREE":
                    player.getOutOfJailFreeCards = (player.getOutOfJailFreeCards || 0) + 1;
                    break;
                case "NEAREST_RAILROAD":
                    const nearestRR = findNearest(player.position, "RAILROAD");
                    if (player.position > nearestRR) {
                        player.money += PASS_GO_MONEY;
                        appendLog(`${player.name} passe par DÉPART et reçoit ${PASS_GO_MONEY}€.`, 'system');
                    }
                    player.position = nearestRR;
                    positionToken(player, players.findIndex(t => t.id === player.id));
                    const rrProp = PROPERTIES[nearestRR];
                    if (rrProp.owner && rrProp.owner !== player.id) {
                        const ownerRailroads = players.find(p => p.id === rrProp.owner).properties.filter(id => PROPERTIES[id].type === 'RAILROAD').length;
                        const rentAmount = rrProp.rent[ownerRailroads - 1] * 2;
                        handlePayment(player, players.find(p => p.id === rrProp.owner), rentAmount);
                    } else {
                         handleLandOnSpace(player, player.position, 0);
                    }
                    return; 
                case "NEAREST_UTILITY":
                    const nearestUtil = findNearest(player.position, "UTILITY");
                    if (player.position > nearestUtil) {
                        player.money += PASS_GO_MONEY;
                        appendLog(`${player.name} passe par DÉPART et reçoit ${PASS_GO_MONEY}€.`, 'system');
                    }
                    player.position = nearestUtil;
                    positionToken(player, players.findIndex(t => t.id === player.id));
                    const utilProp = PROPERTIES[nearestUtil];
                    if (utilProp.owner && utilProp.owner !== player.id) {
                        const die1 = Math.floor(Math.random() * 6) + 1;
                        const die2 = Math.floor(Math.random() * 6) + 1;
                        const roll = die1 + die2;
                        const rentAmount = roll * 10;
                        appendLog(`${player.name} a fait ${roll} et doit payer 10x le montant, soit ${rentAmount}€.`, "roll");
                        handlePayment(player, players.find(p => p.id === utilProp.owner), rentAmount);
                    } else {
                        handleLandOnSpace(player, player.position, 0);
                    }
                    return;
            }
            
            // Show the card message
            showModal('card', {
                title: type === "CHANCE" ? 'Carte CHANCE' : 'Caisse de Communauté',
                text: card.text,
                onClose: () => {
                    // This function will be called when the modal's OK button is clicked
                    switchTurn();
                }
            });

            renderPlayerCards();
        }
    })
    .set(flyingCard, {
        top: pileEl.getBoundingClientRect().top,
        left: pileEl.getBoundingClientRect().left,
    })
    .to(flyingCard, {
        duration: 0.8,
        top: '50%',
        left: '50%',
        transform: 'translate(-50%, -50%) scale(1.5) rotate(0deg)',
        ease: 'power2.inOut'
    })
    .to(flyingCard, {
        duration: 0.3,
        opacity: 0,
        delay: 0.5
    });
};

/**
 * Initiates the bankruptcy process for a player.
 * @param {object} player - The player who is in debt.
 * @param {number} debt - The amount of money the player owes.
 * @param {object|null} creditor - The player or entity to whom the money is owed (null for the bank).
 */
const handleBankruptcy = (player, debt, creditor = null) => {
    Swal.fire({
        title: 'Risque de Faillite !',
        html: `
            <p class="text-lg">Votre solde est négatif. Vous devez lever des fonds pour payer <strong>${debt}€</strong>.</p>
            <p class="mt-2">Le jeu va automatiquement hypothéquer vos propriétés pour couvrir la dette.</p>
        `,
        icon: 'warning',
        iconColor: '#ef4444',
        confirmButtonText: 'Continuer',
        confirmButtonColor: '#ef4444',
        allowOutsideClick: false
    }).then(() => {
        autoMortgage(player, debt, creditor);
    });
};

const autoMortgage = (player, debt, creditor) => {
    const unmortgagedProperties = player.properties.map(id => PROPERTIES[id]).filter(p => !p.mortgaged && p.houses === 0);
    
    for (const prop of unmortgagedProperties) {
        if (player.money < debt) {
            player.money += prop.price / 2;
            prop.mortgaged = true;
            appendLog(`${player.name} hypothèque automatiquement ${prop.name} pour ${prop.price / 2}€.`, 'system');
        } else {
            break; // Debt is covered
        }
    }

    renderPlayerCards();

    if (player.money < debt) {
        // Still bankrupt
        const creditorName = creditor ? creditor.name : 'la Banque';
        Swal.fire({
            title: 'Faillite !',
            text: `Après avoir tout hypothéqué, ${player.name} ne peut toujours pas payer sa dette. Ses actifs sont transférés à ${creditorName}.`,
            icon: 'error',
            allowOutsideClick: false
        }).then(() => {
            // Asset transfer
            player.properties.forEach(propId => {
                const prop = PROPERTIES[propId];
                if (creditor) {
                    creditor.properties.push(propId);
                    prop.owner = creditor.id;
                } else {
                    prop.owner = null; // Return to bank
                    prop.mortgaged = false;
                }
            });
            if(creditor) {
                creditor.money += player.money;
            }

            // Remove player
            const tokenEl = document.getElementById(player.id);
            if (tokenEl) tokenEl.remove();
            
            const remainingPlayers = players.filter(p => p.id !== player.id);
            
            if (remainingPlayers.length === 1) {
                showEndGameScreen(remainingPlayers[0], players);
            } else {
                players = remainingPlayers;
                if (activePlayerIndex >= players.length) activePlayerIndex = 0;
                switchTurn();
            }
        });
    } else {
        // Debt covered, now pay it
        player.money -= debt;
        if (creditor) creditor.money += debt;
        appendLog(`La dette de ${debt}€ a été réglée.`, 'system');
        renderPlayerCards();
    }
};

    const die2 = Math.floor(Math.random() * 6) + 1;
    const roll = die1 + die2;
    const isDouble = die1 === die2;

    appendLog(`${activePlayer.name} lance les dés...`, 'turn');
    messageBox.textContent = "Lancement des dés 3D en cours...";
    diceResultLabelEl.textContent = 'Résultat: ...';

    roll3DDice(die1, die2, () => {
        // On animation complete
        diceResultLabelEl.textContent = `Résultat: ${die1} + ${die2} = ${roll}${isDouble ? ' (Double!)' : ''}`;
        appendLog(`Résultat des dés: ${die1} et ${die2}. Total: ${roll}.`, 'roll');
        messageBox.textContent = `Résultat: ${roll}. Déplacement de ${activePlayer.tokenName}.`;
        
        // Handle doubles for jail/go-to-jail logic
        if (isDouble) {
            doublesCount++;
            if (doublesCount === 3) {
                appendLog(`${activePlayer.name} a fait 3 doubles consécutifs ! Allez en PRISON.`, 'error');
                activePlayer.position = 10;
                activePlayer.inJail = true;
                doublesCount = 0;
                isAnimating = false;
                switchTurn(); // Turn ends immediately
                return;
            }
        } else {
            doublesCount = 0;
        }

        moveToken(activePlayer, roll, isDouble);
    });
};

const moveToken = (playerData, steps, isDouble) => {
    const originalPos = playerData.position;
    let currentPos = originalPos;
    
    highlightTargetCell(originalPos); // Highlight starting cell

    const tl = gsap.timeline({
        onComplete: () => {
            playerData.position = (originalPos + steps) % SPACES;
            positionToken(playerData, players.findIndex(t => t.id === playerData.id)); 
            
            isAnimating = false;
            clearCellHighlights();
            
            // Core game logic is triggered here, and turn switching is decided inside
            handleLandOnSpace(playerData, playerData.position, steps); 
        }
    });

    const tokenEl = document.getElementById(playerData.id);
    const tokenIndex = players.findIndex(t => t.id === playerData.id);
    const offset = TOKEN_OFFSETS[tokenIndex % TOKEN_OFFSETS.length];

    // Loop through each step for the animation
    for (let i = 0; i < steps; i++) {
        let nextPos = (currentPos + 1) % SPACES;
        const nextCell = CELL_POSITIONS[nextPos];
        
        // Check for passing GO
        if (nextPos === 0 && i < steps - 1) { // Only if passing (not landing on)
             playerData.money += PASS_GO_MONEY;
             playerData.stats.moneyEarned += PASS_GO_MONEY;
             appendLog(`${playerData.name} passe par DÉPART et reçoit ${PASS_GO_MONEY}€.`, 'system');
             renderPlayerCards();
        }

        // Animation for one step (300ms per step)
        tl.to(tokenEl, {
            duration: 0.3, 
            x: nextCell.x - TOKEN_HALF_SIZE + offset.dx,
            y: nextCell.y - TOKEN_HALF_SIZE + offset.dy,
            ease: "power1.inOut",
            onUpdate: () => {
                // Highlight current cell during movement
                clearCellHighlights();
                highlightTargetCell(nextPos);
            }
        });
        
        currentPos = nextPos;
    }

    // If it's a double, the turn should NOT switch immediately, but after the action
    if (!isDouble) {
        // rollButton.disabled is set to false in the action handler (handleLandOnSpace or switchTurn)
    }
};

const switchTurn = () => {
    const activePlayer = players[activePlayerIndex];
    
    // If the player made a double and didn't go to jail, they play again
    if (doublesCount > 0 && activePlayer.position !== 10) {
        appendLog(`${activePlayer.name} a fait un double et rejoue.`, 'turn');
        rollButton.disabled = false;
        saveGameState(); // Save state even on extra turns
        return; // Stay on the same player
    }

    // Reset doubles
    doublesCount = 0;

    players[activePlayerIndex].isActive = false;
    players[activePlayerIndex].stats.turnsTaken++;
    activePlayerIndex = (activePlayerIndex + 1) % players.length;
    players[activePlayerIndex].isActive = true;
    
    // Increment turn number only when Player 1 (index 0) starts their turn
    if (activePlayerIndex === 0) {
        currentTurn++;
    }
    
    players.forEach(updateTokenStyle);
    renderPlayerCards(); // Update player cards (active state)
    
    if (players[activePlayerIndex].inJail) {
        appendLog(`Tour ${currentTurn}: ${players[activePlayerIndex].name} est en PRISON.`, 'turn');
        // Will need specific UI buttons for jail actions here
    } else {
        appendLog(`Tour ${currentTurn}: Au tour de ${players[activePlayerIndex].name}.`, 'turn');
    }

    rollButton.disabled = false;
    updateContextualButtons();
    saveGameState(); // Save at the end of every turn
};

const calculateCellPositions = () => {
    for (let i = 0; i < SPACES; i++) {
        const cellId = `cell-${i}`;
        const cell = document.getElementById(cellId);
        
        if (cell) {
            CELL_POSITIONS[i] = {
                x: cell.offsetLeft + cell.offsetWidth / 2,
                y: cell.offsetTop + cell.offsetHeight / 2,
                width: cell.offsetWidth,
                height: cell.offsetHeight
            };
        }
    }
};

const renderTokens = () => {
    players.forEach((playerData, index) => {
        let tokenEl = document.getElementById(playerData.id);
        
        if (!tokenEl) {
            tokenEl = document.createElement('div');
            tokenEl.id = playerData.id;
            tokenEl.className = `token shadow-lg`;
            tokenEl.style.backgroundColor = playerData.color;
            tokenEl.innerHTML = playerData.icon;
            board.appendChild(tokenEl);
        }
        
        positionToken(playerData, index);
        updateTokenStyle(playerData);
    });
};

const positionToken = (playerData, tokenIndex) => {
    const posIndex = playerData.position;
    const pos = CELL_POSITIONS[posIndex];

    if (!pos) return;

    // Determine co-location index
    const tokensOnSquare = players.filter(t => t.position === posIndex);
    const coLocationIndex = tokensOnSquare.findIndex(t => t.id === playerData.id);

    const offset = TOKEN_OFFSETS[coLocationIndex % TOKEN_OFFSETS.length];

    // Center token and apply offset
    gsap.set(document.getElementById(playerData.id), {
        x: pos.x - TOKEN_HALF_SIZE + offset.dx, 
        y: pos.y - TOKEN_HALF_SIZE + offset.dy, 
    });
};

const updateTokenStyle = (playerData) => {
    const tokenEl = document.getElementById(playerData.id);
    if (tokenEl) {
        tokenEl.classList.toggle('token-active', playerData.isActive);
    }
};

const highlightTargetCell = (pos) => {
    const targetCell = document.getElementById(`cell-${pos}`);
    if (targetCell) {
        targetCell.classList.add('pulse');
    }
};

const clearCellHighlights = () => {
    document.querySelectorAll('.property-cell, .corner').forEach(cell => {
        cell.classList.remove('pulse');
    });
};


const handleInitialLayout = () => {
    calculateCellPositions(); 
    players.forEach((player, index) => positionToken(player, index));
    // Resize three.js canvas on layout change
    if (renderer) {
        const container = document.getElementById('dice3dContainer');
        renderer.setSize(container.clientWidth, container.clientHeight);
        camera.aspect = container.clientWidth / container.clientHeight;
        camera.updateProjectionMatrix();
    }
};

const initializeGame = () => {
    // 1. Validate Player Setup
    const playerCount = parseInt(playerCountSelect.value, 10);
    for (let i = 1; i <= playerCount; i++) {
        const setup = playerSetupData[i];
        if (!setup.name || setup.name.trim() === '') {
            Swal.fire('Erreur de Configuration', `Le nom du Joueur ${i} ne peut pas être vide.`, 'error');
            return;
        }
        if (!setup.pawn) {
            Swal.fire('Erreur de Configuration', `Le Joueur ${i} doit choisir un pion.`, 'error');
            return;
        }
        if (!setup.color) {
            Swal.fire('Erreur de Configuration', `Le Joueur ${i} doit choisir une couleur.`, 'error');
            return;
        }
    }

    // 2. Create players array from setup data
    players = []; // Clear the default players
    for (let i = 1; i <= playerCount; i++) {
        const setup = playerSetupData[i];
        players.push({
            id: `player${i}`,
            name: setup.name,
            tokenName: setup.pawn.name,
            color: setup.color, // Now a hex value
            colorText: setup.color, // Use the hex value directly
            icon: `<i class="fas ${setup.pawn.icon}" style="color: #fff;"></i>`,
            position: 0,
            isActive: i === 1, // First player is active
            money: 1500,
            properties: [],
            inJail: false,
            jailTurns: 0,
            stats: {
                rentPaid: 0,
                moneyEarned: 0,
                turnsTaken: 0,
            }
        });
    }
    activePlayerIndex = 0;
    
    // 3. Hide startup menu and show game
    gsap.to(startupMenu, {
        duration: 0.5,
        opacity: 0,
        onComplete: () => {
            startupMenu.classList.add('hidden');
            gameContainer.classList.remove('hidden');
            
            // Initialize the main game engine
            init3DDice();
            setTimeout(() => {
                handleInitialLayout();
                renderTokens();
                renderPlayerCards();
                updateContextualButtons();
                appendLog("Jeu prêt. Cliquez sur 'Lancer les Dés' pour commencer.", 'system');
            }, 100);
        }
    });
};

});

const SAVE_KEY = 'monopolyGameState';

const saveGameState = () => {
    if (typeof(Storage) === "undefined") {
        console.warn("localStorage n'est pas supporté. La partie ne sera pas sauvegardée.");
        return;
    }
    
    // Create a serializable version of the properties, only storing owner, houses, and mortgaged status
    const propertiesToSave = PROPERTIES.map(p => ({
        owner: p.owner,
        houses: p.houses,
        mortgaged: p.mortgaged
    }));

    const gameState = {
        players: players,
        properties: propertiesToSave,
        activePlayerIndex: activePlayerIndex,
        currentTurn: currentTurn,
        gameInProgress: true // Flag to indicate a game is running
    };

    localStorage.setItem(SAVE_KEY, JSON.stringify(gameState));
    appendLog("Partie sauvegardée automatiquement.", 'system');
};

const loadGameState = () => {
    const savedStateJSON = localStorage.getItem(SAVE_KEY);
    if (!savedStateJSON) return false;

    const savedState = JSON.parse(savedStateJSON);

    // Restore players
    players = savedState.players;
    activePlayerIndex = savedState.activePlayerIndex;
    currentTurn = savedState.currentTurn;

    // Restore properties
    savedState.properties.forEach((savedProp, index) => {
        PROPERTIES[index].owner = savedProp.owner;
        PROPERTIES[index].houses = savedProp.houses;
        PROPERTIES[index].mortgaged = savedProp.mortgaged;
    });

    // Hide startup and show game board
    startupMenu.classList.add('hidden');
    gameContainer.classList.remove('hidden');

    // Re-render everything with the loaded data
    init3DDice();
    setTimeout(() => {
        handleInitialLayout();
        renderTokens();
        renderPlayerCards();
        updateContextualButtons();
        // Update cell owner visuals
        PROPERTIES.forEach(prop => {
            if (prop.owner) {
                const player = players.find(p => p.id === prop.owner);
                const cellEl = document.getElementById(`cell-${prop.id}`);
                if (cellEl && player) {
                    cellEl.style.boxShadow = `inset 0 0 0 4px ${player.color}`;
                }
            }
        });
        appendLog("Partie précédente chargée.", 'system');
    }, 100);

    return true;
};

const showEndGameScreen = (winner, allPlayers) => {
    localStorage.removeItem(SAVE_KEY); // Clear the save game data

    const calculateNetWorth = (player) => {
        let propertyValue = player.properties.reduce((acc, propId) => {
            const prop = PROPERTIES[propId];
            let value = prop.price;
            if (prop.houses) {
                value += prop.houses * prop.houseCost;
            }
            if (prop.mortgaged) {
                value /= 2;
            }
            return acc + value;
        }, 0);
        return player.money + propertyValue;
    };

    allPlayers.forEach(p => p.netWorth = calculateNetWorth(p));
    const sortedPlayers = allPlayers.sort((a, b) => b.netWorth - a.netWorth);

    let leaderboardHTML = sortedPlayers.map((p, index) => `
        <div class="p-2 rounded-lg ${index === 0 ? 'bg-yellow-200' : 'bg-gray-100'}">
            <p class="font-bold text-lg">${index + 1}. ${p.name} - ${p.netWorth.toLocaleString('fr-FR')}€</p>
            <p class="text-xs text-gray-600">
                Tours joués: ${p.stats.turnsTaken} | Argent gagné: ${p.stats.moneyEarned}€ | Loyer payé: ${p.stats.rentPaid}€
            </p>
        </div>
    `).join('');

    Swal.fire({
        title: `🏆 ${winner.name} a gagné la partie ! 🏆`,
        html: `
            <div class="text-left space-y-2">
                ${leaderboardHTML}
            </div>
        `,
        confirmButtonText: 'Nouvelle Partie',
        allowOutsideClick: false,
    }).then(() => {
        location.reload();
    });
};
